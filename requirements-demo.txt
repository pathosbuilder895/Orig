# requirements-demo.txt — runtime-only deps for the demo server (run.py --demo).
#
# Excludes:
#   - sentence-transformers / PyTorch   (optional; tier10/tension_arc fall back to TF-IDF)
#   - psycopg2, sqlalchemy, alembic     (demo uses SQLite via store.py, no ORM)
#   - pydantic-settings, email-validator (only needed by production main.py)
#   - python-jose, passlib, bcrypt       (no JWT auth in demo)
#   - slowapi, prometheus-*              (rate-limiting / metrics: prod only)
#   - black, ruff, mypy, pytest-*        (dev / CI tools)

# ── Web framework ──────────────────────────────────────────────────────────────
# fastapi 0.115.0 required starlette<0.39.0, incompatible with the starlette>=1.3.1
# security pin below. fastapi 0.128.8 required starlette<1.0.0,>=0.40.0, which is
# incompatible with that same pin; fastapi>=0.139.0 declares starlette>=0.46.0 with
# no upper bound, so it can resolve alongside starlette>=1.3.1.
fastapi==0.139.0
uvicorn[standard]==0.34.3
pydantic==2.9.2
# python-multipart <0.0.18 had CVE-2024-53981 (header injection) and later CVEs.
# Fixed in 0.0.27.
python-multipart>=0.0.27
# starlette 0.52.1 had 6 known vulnerabilities: PYSEC-2026-161, PYSEC-2026-248,
# PYSEC-2026-249, CVE-2026-48817, CVE-2026-48818. Fixed across 1.0.1-1.3.1. Use >=1.3.1.
starlette>=1.3.1

# ── Numerics / ML (lightweight) ────────────────────────────────────────────────
numpy==2.2.6
# scikit-learn is used by features/tier10.py (TF-IDF fallback) and the Phase 2
# topic resolver. sentence-transformers is optional — code falls back to TF-IDF.
# <1.7: the committed AI-likelihood artifact was trained under 1.6.1 and fails
# validation on newer unpicklers (see requirements.txt). Bump only with a retrain.
scikit-learn>=1.3,<1.7
spacy>=3.7,<4.0
# langdetect powers the Phase 2 language resolver.
langdetect>=1.0.9,<2.0

# ── Document extraction (file-upload endpoints) ────────────────────────────────
python-docx>=1.1,<2.0
# pypdf <6.10.2 had multiple CVEs (path traversal, XML injection, ReDoS).
# pypdf <6.15.0 adds CVE-2026-71852 and CVE-2026-71870. Baseline uploads run
# PdfReader over student-supplied files (original/upload_utils.py), so parser
# CVEs here are reachable from untrusted input, not just theoretical.
pypdf>=6.15.0

# ── Canvas integration ─────────────────────────────────────────────────────────
# canvas/live_import.py makes real Canvas Submissions API calls when
# CANVAS_BASE_URL/CANVAS_API_TOKEN (or a request-supplied token) are set.
httpx>=0.27,<1.0
