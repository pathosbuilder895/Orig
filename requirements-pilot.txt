# requirements-pilot.txt — runtime deps for the PILOT service (run.py --demo
# with ORIGINAL_ENV=pilot hardening).
#
# = requirements-demo.txt + the LTI 1.3 crypto stack + the Postgres stack.
#
# Why a third file (vs requirements.txt): the full production set pulls
# sentence-transformers/PyTorch, which exceeds Render Starter's memory and
# build limits. The pilot runs the same dashboard app as the demo — it just
# additionally needs python-jose/cryptography, which original/lti.py imports
# lazily (LTI launches 500 without them), and the Postgres/SQLAlchemy stack.

-r requirements-demo.txt

# ── LTI 1.3 (original/lti.py) ──────────────────────────────────────────────────
# jose verifies the platform's RS256 id_token against its JWKS; cryptography
# loads the tool's RSA key for /lti/jwks.
python-jose[cryptography]>=3.4.0,<4.0

# ── Off-box backup encryption (scripts/pg_backup_offbox.py) ───────────────────
# The original-pg-backup job Fernet-encrypts every dump. Declared directly so it
# does not depend on python-jose (kept above only for deferred LTI) pulling it
# in; pinned at the version the lock already resolved.
cryptography==50.0.0

# ── Postgres (original/postgres_repository.py, REPO_BACKEND=postgres) ─────────
# Unlike the local demo, the pilot deploy is exactly where REPO_BACKEND is
# actually flipped to postgres (WS-6 P5 cutover) — requirements-demo.txt
# deliberately excludes this stack because the demo never runs Postgres, but
# that exclusion doesn't hold here. Without it, get_repository()'s lazy
# `from .postgres_repository import PostgresRepository` raises ModuleNotFoundError
# on the first request after REPO_BACKEND=postgres is set, even though the
# process boots cleanly — pinned to match requirements.txt's versions.
sqlalchemy==2.0.51
alembic==1.18.5
psycopg2-binary==2.9.13

# ── Error monitoring (original/monitoring.py) ─────────────────────────────────
# Initialised only when SENTRY_DSN is set; without it the SDK is never imported.
# PII and request bodies are never sent (student writing must not leave the box).
sentry-sdk[fastapi]==2.42.0
